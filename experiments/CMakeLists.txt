cmake_minimum_required(VERSION 3.25)
project(m1_probe LANGUAGES CXX)
find_package(Python REQUIRED COMPONENTS Interpreter Development.Module)
execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir OUTPUT_VARIABLE NB_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
find_package(nanobind CONFIG REQUIRED PATHS "${NB_DIR}")
execute_process(COMMAND "${Python_EXECUTABLE}" -c "import mlx; print(mlx.__path__[0])" OUTPUT_VARIABLE MLX_ROOT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
find_package(MLX CONFIG REQUIRED PATHS "${MLX_ROOT}/share/cmake/MLX" NO_DEFAULT_PATH)
nanobind_add_module(m1_probe NB_DOMAIN mlx probe.cpp)
target_link_libraries(m1_probe PRIVATE mlx)
target_compile_features(m1_probe PRIVATE cxx_std_17)
set_target_properties(m1_probe PROPERTIES BUILD_RPATH "${MLX_ROOT}/lib")
